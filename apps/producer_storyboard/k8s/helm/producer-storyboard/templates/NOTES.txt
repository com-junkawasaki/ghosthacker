1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.grpcGo.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "producer-storyboard.fullname" . }}-grpc-go)
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.grpcGo.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "producer-storyboard.fullname" . }}-grpc-go'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "producer-storyboard.fullname" . }}-grpc-go --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.grpcGo.service.port }}
{{- else if contains "ClusterIP" .Values.grpcGo.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "producer-storyboard.name" . }},app.kubernetes.io/instance={{ .Release.Name }},app.kubernetes.io/component=grpc-go" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8081 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8081:$CONTAINER_PORT
{{- end }}

2. Access Temporal UI:
  kubectl port-forward -n temporal-system svc/temporal-ui 8080:8080
  Then visit: http://localhost:8080

3. Access Temporal Frontend (for workers):
  kubectl port-forward -n temporal-system svc/temporal-frontend 7233:7233

4. View logs:
  kubectl logs -f deployment/{{ include "producer-storyboard.fullname" . }}-grpc-go -n {{ .Release.Namespace }}
  kubectl logs -f deployment/{{ include "producer-storyboard.fullname" . }}-temporal-worker -n {{ .Release.Namespace }}
